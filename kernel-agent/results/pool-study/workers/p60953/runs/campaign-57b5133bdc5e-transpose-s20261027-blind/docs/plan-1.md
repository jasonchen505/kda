Kernel transpose_kernel(x_ptr, out_ptr, M, N, stride_xm, stride_xn, stride_om, stride_on, BLOCK_M, BLOCK_N). Grid = (cdiv(M,BM), cdiv(N,BN)). pid_m, pid_n select tile. row_off = pid_m*BM + arange(BM), col_off = pid_n*BN + arange(BN). Load x[row_off, col_off] with mask row<M & col<N. Store to out[col_off, row_off] with mask col<N & row<M. run(x) allocates torch.empty((N,M), dtype, device), launches, returns out.
